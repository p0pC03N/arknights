# 常见算法
常言道Re就是大胆猜测，一眼盯帧，所以对算法的熟悉无比重要。现在特来总结一番。
## Feistel 结构
Feistel 结构（费斯特尔结构）是一种把分组密码做成可逆变换的经典套路。它的厉害之处在于：轮函数 F 不需要可逆，整个加密仍然能解密回来——解密时只要把轮密钥反过来用。

特征：
+ 数据被分成左右两半（两组寄存器/变量）
+ 每轮都有：tmp = L ^ F(R, key)，然后 L = R; R = tmp（或等价 swap）

解密流程：
+ 轮次倒着来
+ 子密钥 K_i 倒序用
+ 同样的 F 继续算

典型 Feistel 分组密码：
+ DES（最经典的 Feistel）
+ Twofish
+ Blowfish
## SMC
SMC 通常指 Self-Modifying Code（自修改代码）：程序在运行时修改自己的指令字节（.text 段或 JIT 缓冲区），从而让你静态反汇编看到的代码和真实执行的代码不一致。CTF/壳/反调试里非常常见。

SMC 的本质是：运行时把某段内存从“不可写代码页”变成“可写”，然后写入新字节（通常是解密后的指令），再切回可执行去跑。
## AES 🤮
AES 的核心是做很多轮（round）可逆的混淆与扩散：
+ 混淆（confusion）：用非线性替换（S-box）把输入字节映射成输出字节，打乱可预测结构。
+ 扩散（diffusion）：通过行移位、列混合把一个字节的影响扩散到更多字节。

### 标准流程（AES-128（10轮））
+ 初始：AddRoundKey（第0轮密钥）

+ 第1~9轮：

  1. SubBytes（S-box字节替换，非线性）

  2. ShiftRows（行循环移位）

  3. MixColumns（列线性混合，GF(2^8) 上的矩阵乘）

  4. AddRoundKey（异或轮密钥）
+ 最后一轮（第10轮）没有 MixColumns：

  SubBytes → ShiftRows → AddRoundKey
### 魔改点
轮数，S盒，key，或者直接变种

```c
// Round constants for AES-128 key expansion (Rcon[i] is exponentiation of 2 in GF(2^8))
static const uint8_t rcon[11] = {
    0x00, // unused (for simpler indexing)
    0x01,0x02,0x04,0x08,0x10,0x20,0x40,0x80,0x1B,0x36
};

static uint8_t xtime(uint8_t x) {
    // multiply by 2 in GF(2^8)
    return (uint8_t)((x << 1) ^ ((x & 0x80) ? 0x1B : 0x00));
}

static void sub_bytes(uint8_t state[16]) {
    for (int i = 0; i < 16; i++) state[i] = sbox[state[i]];
}

static void shift_rows(uint8_t state[16]) {
    uint8_t tmp;

    // Row 1 shift left by 1: (1,5,9,13)
    tmp = state[1];
    state[1]  = state[5];
    state[5]  = state[9];
    state[9]  = state[13];
    state[13] = tmp;

    // Row 2 shift left by 2: (2,6,10,14)
    tmp = state[2];
    state[2]  = state[10];
    state[10] = tmp;
    tmp = state[6];
    state[6]  = state[14];
    state[14] = tmp;

    // Row 3 shift left by 3 (or right by 1): (3,7,11,15)
    tmp = state[15];
    state[15] = state[11];
    state[11] = state[7];
    state[7]  = state[3];
    state[3]  = tmp;
}

static void mix_columns(uint8_t state[16]) {
    for (int c = 0; c < 4; c++) {
        uint8_t *col = &state[c * 4]; // column start (a0,a1,a2,a3)
        uint8_t a0 = col[0], a1 = col[1], a2 = col[2], a3 = col[3];

        uint8_t t  = (uint8_t)(a0 ^ a1 ^ a2 ^ a3);
        uint8_t u0 = a0;

        col[0] = (uint8_t)(a0 ^ t ^ xtime((uint8_t)(a0 ^ a1)));
        col[1] = (uint8_t)(a1 ^ t ^ xtime((uint8_t)(a1 ^ a2)));
        col[2] = (uint8_t)(a2 ^ t ^ xtime((uint8_t)(a2 ^ a3)));
        col[3] = (uint8_t)(a3 ^ t ^ xtime((uint8_t)(a3 ^ u0)));
    }
}

static void add_round_key(uint8_t state[16], const uint8_t round_key[16]) {
    for (int i = 0; i < 16; i++) state[i] ^= round_key[i];
}

static void rot_word(uint8_t w[4]) {
    uint8_t t = w[0];
    w[0] = w[1];
    w[1] = w[2];
    w[2] = w[3];
    w[3] = t;
}

static void sub_word(uint8_t w[4]) {
    w[0] = sbox[w[0]];
    w[1] = sbox[w[1]];
    w[2] = sbox[w[2]];
    w[3] = sbox[w[3]];
}

// Expand 16-byte key into 176-byte expanded key (AES-128)
void AES128_KeyExpansion(uint8_t expanded_key[AES128_EXPKEY_SIZE], const uint8_t key[AES128_KEY_SIZE]) {
    // First round key is the key itself
    for (int i = 0; i < AES128_KEY_SIZE; i++) expanded_key[i] = key[i];

    int bytes_generated = AES128_KEY_SIZE;
    int rcon_iter = 1;
    uint8_t temp[4];

    while (bytes_generated < AES128_EXPKEY_SIZE) {
        // last 4 bytes
        temp[0] = expanded_key[bytes_generated - 4];
        temp[1] = expanded_key[bytes_generated - 3];
        temp[2] = expanded_key[bytes_generated - 2];
        temp[3] = expanded_key[bytes_generated - 1];

        if ((bytes_generated % 16) == 0) {
            rot_word(temp);
            sub_word(temp);
            temp[0] ^= rcon[rcon_iter++];
        }

        // XOR with 16 bytes before
        for (int i = 0; i < 4; i++) {
            expanded_key[bytes_generated] = (uint8_t)(expanded_key[bytes_generated - 16] ^ temp[i]);
            bytes_generated++;
        }
    }
}

// Encrypt one 16-byte block in-place using expanded key
void AES128_ECB_encrypt_block(uint8_t block[16], const uint8_t expanded_key[AES128_EXPKEY_SIZE]) {
    uint8_t state[16];
    for (int i = 0; i < 16; i++) state[i] = block[i];

    // Round 0
    add_round_key(state, &expanded_key[0]);

    // Rounds 1..9
    for (int round = 1; round < AES128_ROUNDS; round++) {
        sub_bytes(state);
        shift_rows(state);
        mix_columns(state);
        add_round_key(state, &expanded_key[round * 16]);
    }

    // Final round (10): no MixColumns
    sub_bytes(state);
    shift_rows(state);
    add_round_key(state, &expanded_key[AES128_ROUNDS * 16]);

    for (int i = 0; i < 16; i++) block[i] = state[i];
}
```

## RC4
主要有两个步骤，KSA&PRGA
+ KSA：初始化一个 256 字节的状态数组 S，内容为 0..255，然后用 key 对它做打乱
+ PRGA：打乱后的 S 产生密钥流
### 标准流程
KSA
+ 初始：S[i] = i，j = 0
+ 对 i=0..255：
  + j = (j + S[i] + key[i mod keylen]) mod 256
  + 交换 S[i] 和 S[j]
PRGA
+ i = 0, j = 0
+ 每输出一个字节：
  + i = (i + 1) mod 256
  + j = (j + S[i]) mod 256
  + 交换 S[i] 和 S[j]
  + t = (S[i] + S[j]) mod 256
  + K = S[t] ← 这就是密钥流字节
  + out = in XOR K
### 魔改点
1. KSA 的 j 更新公式改了
   
  标准：j = j + S[i] + key[i%keylen]

  魔改：

  + j = j + S[i] + key[...] + const
  + j = j ^ S[i] ^ key[...]
  + j = (j + S[i]) ^ key[...]
  + 或 key 索引不是 i%keylen，而是别的（例如 i&7、(i+seed)%keylen）
2. 每字节 XOR 前后再做一层变换
   
  out = (in ^ K) + c 或 out = rol(in ^ K, 3)

  或先对明文做异或/加减，再 RC4，再异或
### 解密代码
```python
from typing import ByteString

def rc4_crypt(key: ByteString, data: ByteString) -> bytes:
    key = bytes(key)
    data = bytes(data)
    if len(key) == 0:
        raise ValueError("key must not be empty")

    # KSA
    S = list(range(256))
    j = 0
    for i in range(256):
        j = (j + S[i] + key[i % len(key)]) & 0xFF
        S[i], S[j] = S[j], S[i]

    # PRGA
    i = 0
    j = 0
    out = bytearray(len(data))
    for n, byte in enumerate(data):
        i = (i + 1) & 0xFF
        j = (j + S[i]) & 0xFF
        S[i], S[j] = S[j], S[i]
        K = S[(S[i] + S[j]) & 0xFF]
        out[n] = byte ^ K

    return bytes(out)

# --- example ---
if __name__ == "__main__":
    key = b"Key"
    ct  = bytes.fromhex("bbf316e8d940af0ad3")  
    pt = rc4_crypt(key, ct)
    print(pt)  # b"Plaintext"
```
## Twofish
AES变种
### 标准流程
1. 128-bit block 拆成 4 个 32-bit word
2. Key Schedule
   40 个 32-bit 子密钥 K[0..39]

   + K[0..3]：输入白化（pre-whitening）
   + K[4..7]：输出白化（post-whitening）
   + K[8..39]：16 轮用的轮密钥（每轮 2 个：K[8+2r]、K[8+2r+1]）
  
   与密钥相关的 S-box 材料（通常记作 S 或 sboxKeys）

   + 由密钥字节通过 RS 矩阵（Reed–Solomon 码相关的固定矩阵）生成
   + 用来构造轮函数里 g() 的 key-dependent S-box（实现上可能是“动态查表”或“预计算大表”）
3. 加密流程
   1. 把明文装载成 x0..x3 后：
    ```
      x0 ^= K[0]
      x1 ^= K[1]
      x2 ^= K[2]
      x3 ^= K[3]
    ```
    2. 16 轮迭代（Feistel + 轮函数）
       1. 计算两个 32-bit 值（轮函数的核心）：
        ```
        t0 = g(x0)          // 依赖 key-dependent S-box + MDS
        t1 = g(ROL(x1, 8))  // 注意这里常见是把 x1 旋转 8 位再进 g
        ```
        2. 做 PHT（Pseudo-Hadamard Transform）并加轮子密钥：
        ```
        F0 = (t0 + t1 + K[8 + 2r]) mod 2^32
        F1 = (t0 + 2*t1 + K[8 + 2r + 1]) mod 2^32
        ```
        3. 混到右半边（并带旋转）：
        ```
        x2 = ROR(x2 ^ F0, 1)
        x3 = ROL(x3, 1) ^ F1
        ```
        4. 交换左右（Feistel 结构的“换位”）
        标准写法通常每轮末尾交换 (x0,x1) 和 (x2,x3)，等价写法很多。最常见表达是：
        ```
        swap(x0, x2)
        swap(x1, x3)
        ```
4. g() 函数
   
   输入 32-bit，输出 32-bit。
    1. 把输入 32-bit 拆成 4 个字节 b0,b1,b2,b3
    2. 对每个字节做若干层 q 置换（q0/q1 是固定 8×8 置换），中间夹杂与 S 相关的 XOR
    3. 得到 4 个字节后，乘以固定的 4×4 MDS 矩阵（GF(2^8)）扩散成 32-bit 输出
   
    会看到 q0/q1 两张 256 表（或其拆分/混淆版本）+ MDS 乘法（或预计算的 MDS 表）。
### 魔改点
g()函数，q0/q1 两张 256 表+ MDS表
### 解密代码
1. x0..x3 从密文装载
2. 先 XOR K[4..7]（undo post-whitening）
3. 16 轮逆序：swap 的位置按实现对应；轮函数 g() 本身仍然是正向的
4. 最后 XOR K[0..3]（undo pre-whitening）
5. 输出明文
```python
# -*- coding: utf-8 -*-
import struct

MASK32 = 0xFFFFFFFF

def rol32(x, r):
    r &= 31
    x &= MASK32
    return ((x << r) & MASK32) | (x >> (32 - r) if r else 0)

def ror32(x, r):
    r &= 31
    x &= MASK32
    return (x >> r) | ((x << (32 - r)) & MASK32 if r else 0)

# -------------------------
# q0/q1 (Twofish fixed 8-bit permutations) built from 4-bit tables
# -------------------------
_q0 = [
    [0x8,0x1,0x7,0xD,0x6,0xF,0x3,0x2,0x0,0xB,0x5,0x9,0xE,0xC,0xA,0x4],
    [0xE,0xC,0xB,0x8,0x1,0x2,0x3,0x5,0xF,0x4,0xA,0x6,0x7,0x0,0x9,0xD],
    [0xB,0xA,0x5,0xE,0x6,0xD,0x9,0x0,0xC,0x8,0xF,0x3,0x2,0x4,0x7,0x1],
    [0xD,0x7,0xF,0x4,0x1,0x2,0x6,0xE,0x9,0xB,0x3,0x0,0x8,0x5,0xC,0xA],
]
_q1 = [
    [0x2,0x8,0xB,0xD,0xF,0x7,0x6,0xE,0x3,0x1,0x9,0x4,0x0,0xA,0xC,0x5],
    [0x1,0xE,0x2,0xB,0x4,0xC,0x3,0x7,0x6,0xD,0xA,0x5,0xF,0x9,0x0,0x8],
    [0x4,0xC,0x7,0x5,0x1,0x6,0x9,0xA,0x0,0xE,0xD,0x8,0x2,0xB,0x3,0xF],
    [0xB,0x9,0x5,0x1,0xC,0x3,0xD,0xE,0x6,0x4,0x7,0xF,0x2,0x0,0x8,0xA],
]

def _ror4(x, r):
    r &= 3
    x &= 0xF
    return ((x >> r) | ((x << (4-r)) & 0xF)) & 0xF

def _qperm(which, x):
    a0 = (x >> 4) & 0xF
    b0 = x & 0xF

    if which == 0:
        a1 = a0 ^ b0
        b1 = a0 ^ _ror4(b0, 1) ^ ((a0 << 3) & 0xF)
        a2 = _q0[0][a1]; b2 = _q0[1][b1]
        a3 = a2 ^ b2
        b3 = a2 ^ _ror4(b2, 1) ^ ((a2 << 3) & 0xF)
        a4 = _q0[2][a3]; b4 = _q0[3][b3]
    else:
        a1 = a0 ^ b0
        b1 = a0 ^ _ror4(b0, 1) ^ ((a0 << 3) & 0xF)
        a2 = _q1[0][a1]; b2 = _q1[1][b1]
        a3 = a2 ^ b2
        b3 = a2 ^ _ror4(b2, 1) ^ ((a2 << 3) & 0xF)
        a4 = _q1[2][a3]; b4 = _q1[3][b3]

    return ((b4 << 4) | a4) & 0xFF

Q0 = [ _qperm(0,i) for i in range(256) ]
Q1 = [ _qperm(1,i) for i in range(256) ]

# -------------------------
# GF(2^8) mul with Twofish MDS polynomial 0x169
# -------------------------
def gf_mul(a, b, poly_low=0x69):
    a &= 0xFF; b &= 0xFF
    res = 0
    for _ in range(8):
        if b & 1:
            res ^= a
        b >>= 1
        carry = a & 0x80
        a = (a << 1) & 0xFF
        if carry:
            a ^= poly_low
    return res & 0xFF

def _m1(x): return x & 0xFF
def _mX(x): return gf_mul(x, 0x5B)   # standard Twofish multiplier
def _mY(x): return gf_mul(x, 0xEF)

def build_mds_tables():
    T = [[0]*256 for _ in range(4)]
    for i in range(256):
        m1, mX, mY = _m1(i), _mX(i), _mY(i)
        # pack into u32 (little-endian bytes)
        T[0][i] = (m1) | (mX<<8) | (mY<<16) | (mY<<24)
        T[1][i] = (mY) | (mY<<8) | (mX<<16) | (m1<<24)
        T[2][i] = (mX) | (mY<<8) | (m1<<16) | (mY<<24)
        T[3][i] = (mX) | (m1<<8) | (mY<<16) | (mX<<24)
    return T

MDS = build_mds_tables()

# -------------------------
# RS encode (standard Twofish, GF poly 0x14D) -> S-box key words
# -------------------------
def gf_mul_rs(a, b, poly_low=0x4D):
    a &= 0xFF; b &= 0xFF
    res = 0
    for _ in range(8):
        if b & 1:
            res ^= a
        b >>= 1
        carry = a & 0x80
        a = (a << 1) & 0xFF
        if carry:
            a ^= poly_low
    return res & 0xFF

RS = [
    [0x01,0xA4,0x55,0x87,0x5A,0x58,0xDB,0x9E],
    [0xA4,0x56,0x82,0xF3,0x1E,0xC6,0x68,0xE5],
    [0x02,0xA1,0xFC,0xC1,0x47,0xAE,0x3D,0x19],
    [0xA4,0x55,0x87,0x5A,0x58,0xDB,0x9E,0x03],  # 常见实现用这个版本
]

def rs_mds_encode(k0, k1):
    b = list(struct.pack("<2I", k0 & MASK32, k1 & MASK32))  # 8 bytes
    out = []
    for r in range(4):
        acc = 0
        for c in range(8):
            acc ^= gf_mul_rs(b[c], RS[r][c])
        out.append(acc & 0xFF)
    return struct.unpack("<I", bytes(out))[0] & MASK32

# -------------------------
# h(x, L) for Twofish-128  (key-dependent S-box layers)
# For 128-bit key: 2 layers
# -------------------------
def h128(x, L_words):
    b0 = x & 0xFF
    b1 = (x >> 8) & 0xFF
    b2 = (x >> 16) & 0xFF
    b3 = (x >> 24) & 0xFF

    # L_words is two 32-bit words -> 8 bytes (little-endian)
    L = list(struct.pack("<2I", L_words[0] & MASK32, L_words[1] & MASK32))

    # layer 1
    b0 = Q0[b0] ^ L[0]
    b1 = Q1[b1] ^ L[1]
    b2 = Q0[b2] ^ L[2]
    b3 = Q1[b3] ^ L[3]
    # layer 2
    b0 = Q0[b0] ^ L[4]
    b1 = Q0[b1] ^ L[5]
    b2 = Q1[b2] ^ L[6]
    b3 = Q1[b3] ^ L[7]

    return (MDS[0][b0] ^ MDS[1][b1] ^ MDS[2][b2] ^ MDS[3][b3]) & MASK32

def make_subkeys_and_sboxkeys(key16: bytes):
    if len(key16) != 16:
        raise ValueError("This minimal code supports Twofish-128 only (16-byte key).")

    k = list(struct.unpack("<4I", key16))
    Me = [k[0], k[2]]
    Mo = [k[1], k[3]]

    # S-box keys (2 words)
    S = [rs_mds_encode(Me[0], Mo[0]), rs_mds_encode(Me[1], Mo[1])]

    # 40 subkeys
    K = [0]*40
    for i in range(20):
        A = h128((2*i) & MASK32, Me)
        B = rol32(h128((2*i+1) & MASK32, Mo), 8)
        K[2*i]   = (A + B) & MASK32
        K[2*i+1] = rol32((A + 2*B) & MASK32, 9)  # standard is 9
    return K, S

def g128(x, S_words):
    # In standard Twofish, g(x) is h(x, S) where S are the RS-derived words.
    return h128(x, S_words)

# -------------------------
# Minimal decrypt using K and g(x) (standard rotations)
# -------------------------
def twofish128_decrypt_block(ct16: bytes, K, S_words) -> bytes:
    x0, x1, x2, x3 = struct.unpack("<4I", ct16)

    # undo output whitening
    x0 ^= K[4]; x1 ^= K[5]; x2 ^= K[6]; x3 ^= K[7]

    for r in range(15, -1, -1):
        # undo swap
        x0, x2 = x2, x0
        x1, x3 = x3, x1

        t0 = g128(x0, S_words)
        t1 = g128(rol32(x1, 8), S_words)
        F0 = (t0 + t1 + K[8 + 2*r]) & MASK32
        F1 = (t0 + 2*t1 + K[8 + 2*r + 1]) & MASK32

        # inverse of standard encryption:
        # x2 = ror(x2 ^ F0, 1)  => prev_x2 = rol(x2, 1) ^ F0
        # x3 = rol(x3, 1) ^ F1  => prev_x3 = ror(x3 ^ F1, 1)
        x2 = rol32(x2, 1) ^ F0
        x3 = ror32(x3 ^ F1, 1)

    # undo input whitening
    x0 ^= K[0]; x1 ^= K[1]; x2 ^= K[2]; x3 ^= K[3]
    return struct.pack("<4I", x0 & MASK32, x1 & MASK32, x2 & MASK32, x3 & MASK32)

# (可选) 加密用于自测：加密->解密应回原文
def twofish128_encrypt_block(pt16: bytes, K, S_words) -> bytes:
    x0, x1, x2, x3 = struct.unpack("<4I", pt16)

    # input whitening
    x0 ^= K[0]; x1 ^= K[1]; x2 ^= K[2]; x3 ^= K[3]

    for r in range(16):
        t0 = g128(x0, S_words)
        t1 = g128(rol32(x1, 8), S_words)
        F0 = (t0 + t1 + K[8 + 2*r]) & MASK32
        F1 = (t0 + 2*t1 + K[8 + 2*r + 1]) & MASK32

        x2 = ror32(x2 ^ F0, 1)
        x3 = rol32(x3, 1) ^ F1

        # swap
        x0, x2 = x2, x0
        x1, x3 = x3, x1

    # undo final swap
    x0, x2 = x2, x0
    x1, x3 = x3, x1

    # output whitening
    x0 ^= K[4]; x1 ^= K[5]; x2 ^= K[6]; x3 ^= K[7]
    return struct.pack("<4I", x0 & MASK32, x1 & MASK32, x2 & MASK32, x3 & MASK32)

if __name__ == "__main__":
    key = b"WONDERFUL&&PEACE"  # 16 bytes
    K, S = make_subkeys_and_sboxkeys(key)

    pt = b"0123456789ABCDEF"
    ct = twofish128_encrypt_block(pt, K, S)
    rt = twofish128_decrypt_block(ct, K, S)
    assert rt == pt, (ct.hex(), rt)
    print("OK. ct =", ct.hex())
```

## DES
对称密钥分组密码，典型 Feistel 网络

1. 初始置换 IP：M' = IP(M)
2. 切成左右 32-bit：L0 || R0
3. 做 16 轮：
   + Li = Ri-1
   + Ri = Li-1 XOR F(Ri-1, Ki)
4. 交换（有的描述把最后交换吸收在最终置换里）：R16 || L16
5. 逆初始置换 FP（IP^-1）：输出密文 C = FP(R16 || L16)
### 标准流程
1. 轮函数 F 的组成（核心）：
   F(R, K) 输入 32-bit 的 R 和 48-bit 子密钥 K，输出 32-bit：
   1. 扩展置换 E：把 R(32) 扩展为 E(R)(48)（有重复位）
   2. 异或子密钥：B = E(R) XOR K（48 bit）
   3. S-box 代换：把 48-bit 分成 8 组，每组 6-bit 进一个 S-box，输出 4-bit
    + 8 个 S-box：8 × (6->4) 合成 32-bit
    + 取行列：行由第1和第6位，列由中间4位
   4. 置换 P：对 32-bit 做固定置换扩散
2. 子密钥生成（Key Schedule）
    主密钥（含校验位）先经过 PC-1 选出 56 bit，分成 C0(28) 与 D0(28)

    每轮左移（1或2位，按轮数表）得到 Ci, Di

    再经过 PC-2 压缩选择成 48-bit 子密钥 Ki（i=1..16）
### 魔改点
1. 轮数
2. S-box
3. E 表（48项）和 P 表（32项）
4. Key schedule

```python
# -------- helpers --------
def permute_bits(x: int, in_bits: int, table: list[int]) -> int:
    """table is 1-based positions, MSB-first in_bits."""
    out = 0
    for pos in table:
        out = (out << 1) | ((x >> (in_bits - pos)) & 1)
    return out

def rol28(x: int, r: int) -> int:
    x &= 0x0FFFFFFF
    return ((x << r) | (x >> (28 - r))) & 0x0FFFFFFF

def expand32to48(r: int) -> int:
    out = 0
    for pos in E:
        out = (out << 1) | ((r >> (32 - pos)) & 1)
    return out  # 48-bit

def sbox_sub(x48: int) -> int:
    out32 = 0
    for b in range(8):
        six = (x48 >> (42 - 6*b)) & 0x3F
        row = ((six & 0x20) >> 4) | (six & 0x01)
        col = (six >> 1) & 0x0F
        out32 = (out32 << 4) | SBOX[b][row*16 + col]
    return out32

def f_func(r: int, k48: int) -> int:
    x = expand32to48(r) ^ (k48 & 0xFFFFFFFFFFFF)
    s = sbox_sub(x & 0xFFFFFFFFFFFF)
    return permute_bits(s, 32, P)

# -------- key schedule --------
def des_key_schedule(key8: bytes) -> list[int]:
    if len(key8) != 8:
        raise ValueError("DES key must be 8 bytes (64 bits incl parity).")

    k64 = int.from_bytes(key8, "big")
    k56 = permute_bits(k64, 64, PC1)  # 56 bits

    C = (k56 >> 28) & 0x0FFFFFFF
    D = k56 & 0x0FFFFFFF

    subkeys = []
    for r in range(16):
        C = rol28(C, SHIFTS[r])
        D = rol28(D, SHIFTS[r])
        cd56 = (C << 28) | D
        k48 = permute_bits(cd56, 56, PC2) & 0xFFFFFFFFFFFF
        subkeys.append(k48)
    return subkeys

# -------- block encrypt/decrypt --------
def des_encrypt_block(pt8: bytes, subkeys: list[int]) -> bytes:
    if len(pt8) != 8:
        raise ValueError("block must be 8 bytes")
    x = int.from_bytes(pt8, "big")
    ip = permute_bits(x, 64, IP)
    L = (ip >> 32) & 0xFFFFFFFF
    R = ip & 0xFFFFFFFF

    for r in range(16):
        L, R = R, (L ^ f_func(R, subkeys[r])) & 0xFFFFFFFF

    preout = ((R & 0xFFFFFFFF) << 32) | (L & 0xFFFFFFFF)  # final swap
    fp = permute_bits(preout, 64, FP)
    return fp.to_bytes(8, "big")

def des_decrypt_block(ct8: bytes, subkeys: list[int]) -> bytes:
    if len(ct8) != 8:
        raise ValueError("block must be 8 bytes")
    x = int.from_bytes(ct8, "big")
    ip = permute_bits(x, 64, IP)
    L = (ip >> 32) & 0xFFFFFFFF
    R = ip & 0xFFFFFFFF

    for r in range(16):
        L, R = R, (L ^ f_func(R, subkeys[15 - r])) & 0xFFFFFFFF  # reverse keys

    preout = ((R & 0xFFFFFFFF) << 32) | (L & 0xFFFFFFFF)
    fp = permute_bits(preout, 64, FP)
    return fp.to_bytes(8, "big")
```
